\newglsgroup{general}{General}

\glsxtrnewsymbol[description={arbitrary (scalar) variable}, group=general]
  {arbitrary_scalar}{\ensuremath{(\cdot)}}
\glsxtrnewsymbol[description={arbitrary vector variable}, group=general]
  {arbitrary_vector}{\ensuremath{(\bullet)}}
\glsxtrnewsymbol[description={approximately equal}, group=general]
  {approximately_equal}{\ensuremath{\approx}}
\glsxtrnewsymbol[description={tuple of multiple variables}, group=general]
  {tuple}{\ensuremath{\tuple{\cdot, \cdot, \cdot}}}
\glsxtrnewsymbol[description={set of real numbers}, group=general]
  {real_numbers}{\ensuremath{\symbb{R}}}
\glsxtrnewsymbol[description={expected value of a \ac{rv} $X$}, group=general]
  {expectation}{\ensuremath{\expectation{X}}}
\glsxtrnewsymbol[description={probability that an \ac{rv} $X$ takes on the value $x$}, group=general]
  {probability}{\ensuremath{\probability{X=x}}}
\glsxtrnewsymbol[description={partial derivative of $f$ with respect to $x$}, group=general]
  {partial_derivative}{\ensuremath{\frac{\partial f(\bullet)}{\partial x}}}
\glsxtrnewsymbol[description={partial derivative of $f$ with respect to $x$ evaluated at $\bvec{a}$}, group=general]
  {evaluated_at}{\ensuremath{\evalat[\bvec{a}] {\frac{\partial f(\bullet)}{\partial x}}}}
\glsxtrnewsymbol[description={optimal value of $a$ minimizing the expression $f(a)$}, group=general]
  {argmin}{\ensuremath{\argmin_a f(a)}}
\glsxtrnewsymbol[description={optimal value of $a$ maximizing the expression $f(a)$}, group=general]
  {argmax}{\ensuremath{\argmax_a f(a)}}
\glsxtrnewsymbol[description={mathematical constant, $\uppi \approx 3.142$}, group=general]
  {pi_constant}{\ensuremath{\uppi}}
\glsxtrnewsymbol[description={Gaussian normal distribution with mean~$\mu$ and standard deviation~$\sigma$}, group=general]
  {normal_distribution}{\ensuremath{\symcal{N}(\mu, \sigma)}}


\newglsgroup{mdp}{Markov Decision Process}

\glsxtrnewsymbol[description={time step index}, group=mdp]
  {time_step}{\ensuremath{t}}
\glsxtrnewsymbol[description={set of all states}, group=mdp]
  {state_space}{\ensuremath{\symcal{S}}}
\glsxtrnewsymbol[description={set of all actions}, group=mdp]
  {action_space}{\ensuremath{\symcal{A}}}
\glsxtrnewsymbol[description={set of all possible rewards}, group=mdp]
  {reward_space}{\ensuremath{\symcal{R}}}
\glsxtrnewsymbol[description={state at time $t$, realization of \ac{rv} $S^t$}, group=mdp]
  {state}{\ensuremath{s}, \ensuremath{s^t}}
\glsxtrnewsymbol[description={next state at time $t+1$, realization of \ac{rv} $S^{t+1}$}, group=mdp]
  {next_state}{\ensuremath{s'}, \ensuremath{s^{t+1}}}
\glsxtrnewsymbol[description={action at time $t$, realization of \ac{rv} $A^t$}, group=mdp]
  {action}{\ensuremath{a}, \ensuremath{a^t}}
\glsxtrnewsymbol[description={next action at time $t+1$, realization of \ac{rv} $A^{t+1}$}, group=mdp]
  {next_action}{\ensuremath{a'}, \ensuremath{a^{t+1}}}
\glsxtrnewsymbol[description={reward at time $t+1$, realization of \ac{rv} $R^{t+1}$}, group=mdp]
  {reward}{\ensuremath{r}, \ensuremath{r^{t+1}}}
\glsxtrnewsymbol[description={state at time $t$}, group=mdp]
  {state_rv}{\ensuremath{S^t}}
\glsxtrnewsymbol[description={action at time $t$}, group=mdp]
  {action_rv}{\ensuremath{A^t}}
\glsxtrnewsymbol[description={reward at time $t$}, group=mdp]
  {reward_rv}{\ensuremath{R^t}}
\glsxtrnewsymbol[description={dynamics function specifying a conditional \ac{pdf} for the next state~$s'$ and reward~$r$ given the current state~$s$ and action~$a$}, group=mdp]
  {dynamics_function}{\ensuremath{p(s', r \given s, a)}}
